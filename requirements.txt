# The TFLite runtime is NOT listed here — it is target-specific:
#
#   Raspberry Pi 4   pip install ai-edge-litert       (setup.sh does this)
#                    tflite-runtime has no wheel for Python 3.13, which
#                    Raspberry Pi OS 13 (Trixie) ships; ai-edge-litert is
#                    its successor after TFLite was renamed LiteRT.
#   Coral (legacy)   sudo apt-get install -y python3-pycoral libedgetpu1-std
#                    (Mendel 5 also ships python3-tflite-runtime 2.5.0)
#
# Floors only, no ceilings. On the Pi, pip resolves these to current
# wheels — on 2026-11-25 that was numpy 2.5.5 and OpenCV 5.1, and 11 of
# the 12 test files passed on them (the 12th, test_camera_sim.py, needs
# pytest, which is not installed on the board). The floors are as low as they are because
# they were set for the legacy Coral (Mendel 5: Debian 10, Python 3.7,
# numpy 1.19, OpenCV 4.5.4), where demanding newer versions forced long
# source builds. They are harmless on the Pi, so they stay.
#
opencv-python-headless>=4.5.4
numpy>=1.16.0
PyYAML>=5.4
flask>=2.2.5
twilio>=8.10.0

# Pinned below 2.0: paho-mqtt 2.x requires a CallbackAPIVersion argument to
# Client(), which alerts/mqtt_client.py does not pass. Version 2.x raises
# ValueError at construction, which the MQTT init swallows — silently
# disabling MQTT rather than failing visibly.
paho-mqtt>=1.6.1,<2.0
